2-lik (Binary)

Binary → Text:	echo "01101000 01100001 01101001" | python3 -c 'import sys; print("".join(chr(int(x,2)) for x in sys.stdin.read().split()))'
Text → Binary:	echo -n "haiCTF" | python3 -c 'import sys; print(" ".join(format(ord(x),"08b") for x in sys.stdin.read()))'


8-lik (Octal)

Octal → Text:	echo "150 141 151" | python3 -c 'import sys; print("".join(chr(int(x,8)) for x in sys.stdin.read().split()))'
Text → Octal:	echo -n "haiCTF" | python3 -c 'import sys; print(" ".join(format(ord(x),"o") for x in sys.stdin.read()))'


10-lik (Decimal)

Decimal → Text:	echo "104 97 105" | python3 -c 'import sys; print("".join(chr(int(x)) for x in sys.stdin.read().split()))'
Text → Decimal:	echo -n "haiCTF" | python3 -c 'import sys; print(" ".join(str(ord(x)) for x in sys.stdin.read()))'


16-lik (Hex)

Hex → Text:	echo "686169435446" | xxd -r -p
Text → Hex:	echo -n "haiCTF" | xxd -p

Text → Base64:	echo -n "hello" | base64
